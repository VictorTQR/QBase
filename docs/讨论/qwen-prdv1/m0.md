```python
model = ""
dimension = 0
batch_size = 32
timeout = 120

[index]

chunk_overlap = 100
rebuild_batch_size = 100
"""


def ensure_knowledge_dirs(root: Path) -> Path:
    """
    在知识库根目录下创建 .knowledge 应用数据目录。
    """
    knowledge_dir = root / ".knowledge"

    dirs = [
        knowledge_dir,
        knowledge_dir / "vector" / "lancedb",
        knowledge_dir / "cache",
        knowledge_dir / "backups",
        knowledge_dir / "logs",
        knowledge_dir / "tasks",
    ]

    for path in dirs:
        path.mkdir(parents=True, exist_ok=True)

    config_path = knowledge_dir / "config.toml"
    if not config_path.exists():
        config_path.write_text(DEFAULT_CONFIG_TOML.strip(), encoding="utf-8")

    return knowledge_dir
```

## app/database.py

初始化 SQLite 数据库。

```python
import sqlite3
from pathlib import Path


SCHEMA = """
CREATE TABLE IF NOT EXISTS assets (
  id TEXT PRIMARY KEY,
  title TEXT NOT NULL,
  type TEXT NOT NULL,
  relative_path TEXT NOT NULL UNIQUE,
  absolute_path TEXT NOT NULL,
  mime_type TEXT,
  size INTEGER,
  mtime INTEGER,
  file_hash TEXT,
  duration_seconds REAL,
  parse_status TEXT DEFAULT 'unknown',
  created_at TEXT NOT NULL,
  updated_at TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS artifacts (
  id TEXT PRIMARY KEY,
  asset_id TEXT NOT NULL,
  kind TEXT NOT NULL,
  relative_path TEXT NOT NULL UNIQUE,
  absolute_path TEXT NOT NULL,
  file_hash TEXT,
  mtime INTEGER,
  source TEXT,
  generator TEXT,
  model TEXT,
  status TEXT DEFAULT 'active',
  created_at TEXT NOT NULL,
  updated_at TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS tasks (
  id TEXT PRIMARY KEY,
  asset_id TEXT,
  type TEXT NOT NULL,
  status TEXT NOT NULL,
  command TEXT,
  params_json TEXT,
  output_path TEXT,
  error TEXT,
  pid INTEGER,
  created_at TEXT NOT NULL,
  started_at TEXT,
  finished_at TEXT
);

CREATE TABLE IF NOT EXISTS chunks (
  id TEXT PRIMARY KEY,
  asset_id TEXT NOT NULL,
  artifact_id TEXT,
  kind TEXT NOT NULL,
  relative_path TEXT NOT NULL,
  chunk_index INTEGER NOT NULL,
  content TEXT NOT NULL,
  content_hash TEXT,
  created_at TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS embedding_cache (
  content_hash TEXT PRIMARY KEY,
  model TEXT NOT NULL,
  vector BLOB NOT NULL,
  dimension INTEGER NOT NULL,
  created_at TEXT NOT NULL
);
"""


def init_db(db_path: Path) -> None:
    """
    初始化 SQLite 数据库。
    """
    conn = sqlite3.connect(db_path)
    try:
        conn.executescript(SCHEMA)

        try:
            conn.execute(
                """
                CREATE VIRTUAL TABLE IF NOT EXISTS chunks_fts USING fts5(
                  content,
                  content='chunks',
                  content_rowid='rowid'
                );
                """
            )
        except sqlite3.OperationalError:
            # 某些 SQLite 编译版本可能没有 FTS5，第一阶段先允许跳过。
            pass

        conn.commit()
    finally:
        conn.close()


def get_conn(db_path: Path) -> sqlite3.Connection:
    """
    获取 SQLite 连接。
    """
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys=ON;")
    return conn
```

## app/services/library_service.py

打开或新建知识库。

```python
from pathlib import Path

from app.config import ensure_knowledge_dirs
from app.database import init_db
from app.state import state


def open_library(root_path: str) -> dict:
    """
    打开或新建一个知识库目录。
    """
    if not root_path or not root_path.strip():
        raise ValueError("请输入知识库目录路径")

    root = Path(root_path.strip()).expanduser()

    try:
        root = root.resolve()
    except Exception as exc:
        raise ValueError(f"无法解析路径：{root_path}") from exc

    root.mkdir(parents=True, exist_ok=True)

    if not root.is_dir():
        raise ValueError(f"路径不是目录：{root}")

    knowledge_dir = ensure_knowledge_dirs(root)
    db_path = knowledge_dir / "db.sqlite"

    init_db(db_path)

    state.library_root = root

    return {
        "status": "ok",
        "library_root": str(root),
        "knowledge_dir": str(knowledge_dir),
        "db_path": str(db_path),
    }


def get_library_status() -> dict:
    """
    获取当前知识库状态。
    """
    if state.library_root is None:
        return {
            "opened": False,
            "library_root": None,
        }

    knowledge_dir = state.library_root / ".knowledge"
    db_path = knowledge_dir / "db.sqlite"

    return {
        "opened": True,
        "library_root": str(state.library_root),
        "knowledge_dir": str(knowledge_dir),
        "db_path": str(db_path),
        "db_exists": db_path.exists(),
    }
```

## app/api/library.py

提供最小 API。

```python
from pydantic import BaseModel

from app.services.library_service import open_library, get_library_status


class OpenLibraryRequest(BaseModel):
    path: str


def register_library_api(app):
    """
    将 API 注册到 NiceGUI 内置 FastAPI app。
    """

    @app.get("/api/health")
    def health():
        return {
            "status": "ok",
            "service": "local-knowledge-hub",
        }

    @app.post("/api/library/open")
    def api_open_library(req: OpenLibraryRequest):
        try:
            return open_library(req.path)
        except ValueError as exc:
            return {
                "status": "error",
                "message": str(exc),
            }

    @app.get("/api/library/status")
    def api_library_status():
        return get_library_status()
```

## app/ui/pages/home.py

首页：打开知识库。

```python
from nicegui import ui, run

from app.services.library_service import open_library, get_library_status


@ui.page("/")
def home_page():
    ui.label("本地知识管理中心").classes("text-3xl font-bold")
    ui.label("Local Knowledge Hub").classes("text-sm text-gray-500 mb-4")

    with ui.card().classes("w-[560px] p-5"):
        ui.label("打开或新建知识库").classes("text-lg font-semibold")
        ui.label(
            "输入一个本地目录。应用会在该目录下创建 .knowledge 文件夹。"
        ).classes("text-sm text-gray-600")

        path_input = ui.input(
            label="知识库目录",
            placeholder="例如：D:/Knowledge",
        ).classes("w-full mt-3")

        status_label = ui.label("当前未打开知识库").classes(
            "text-sm text-gray-600 mt-3"
        )

        def refresh_status():
            status = get_library_status()
            if status.get("opened"):
                status_label.text = f"当前知识库：{status['library_root']}"
            else:
                status_label.text = "当前未打开知识库"

        async def handle_open_library():
            try:
                result = await run.io_bound(open_library, path_input.value)
                ui.notify(f"已打开：{result['library_root']}", type="positive")
                refresh_status()
            except Exception as exc:
                ui.notify(str(exc), type="negative")

        ui.button(
            "打开知识库",
            on_click=handle_open_library,
        ).classes("mt-4")

        refresh_status()
```
