            ui.badge("未打开知识库", color="orange").classes("text-xs")


def breadcrumb(items: list[tuple[str, str | None]]):
    """
    面包屑导航。
    items: [(显示文字, 链接路径), ...]，最后一项路径为 None 表示当前页。
    示例：breadcrumb([("首页", "/"), ("资产列表", "/assets"), ("某文件", None)])
    """
    with ui.row().classes("px-4 py-1 items-center gap-1 text-sm text-gray-500"):
        for i, (text, path) in enumerate(items):
            if i > 0:
                ui.label("/").classes("text-gray-400")
            if path is not None:
                ui.link(text, path).classes("text-blue-600 hover:underline")
            else:
                ui.label(text).classes("text-gray-700 font-medium truncate").props(
                    "style='max-width: 200px'"
                )


def require_library():
    """
    检查是否已打开知识库。
    未打开时显示提示并返回 False。

        ui.label("未打开知识库").classes("text-xl mt-6")
        ui.link("去打开知识库", "/").classes("text-blue-600 mt-2")

```python
import html
import re
from pathlib import Path

from loguru import logger


# ──────────────────────────────────────────────
# 错误提示统一层
# ──────────────────────────────────────────────

# 已知异常的中文映射（key 为异常类名字符串，value 为提示模板）
_ERROR_MESSAGES = {
    "FileNotFoundError": "文件不存在：{detail}",
    "PermissionError": "权限不足：{detail}",
    "ValueError": "{detail}",
    "ConnectionError": "网络连接失败，请检查服务是否启动：{detail}",
    "TimeoutError": "操作超时，请稍后重试：{detail}",
}


def notify_error(exc: Exception):
    """
    统一错误处理：
    1. 用 loguru 记录完整堆栈到日志
    2. 对用户显示友好的中文提示
    3. 返回用户友好的消息字符串（供调用方自定义 notify 时使用）
    """
    from nicegui import ui

    # 记录完整堆栈
    logger.exception(f"操作失败：{type(exc).__name__}: {exc}")

    exc_name = type(exc).__name__
    detail = str(exc)

    # 特殊处理：SQLite 操作错误 → 提示索引问题
    if exc_name == "OperationalError" and "no such table" in detail.lower():
        message = "索引可能未建立，请前往设置页重建索引。"
    elif exc_name in _ERROR_MESSAGES:
        message = _ERROR_MESSAGES[exc_name].format(detail=detail)
    else:
        message = f"操作失败：{detail}。详见日志。"

    ui.notify(message, type="negative", multi_line=True)
    return message


# ──────────────────────────────────────────────
# 搜索高亮
# ──────────────────────────────────────────────

def highlight_snippet(snippet: str, words: list[str]) -> str:
    """
    对 snippet 中的命中词进行高亮，返回安全的 HTML 字符串。
    1. 先 HTML 转义
    2. 对每个 word 做大小写不敏感替换，包裹 <mark>
    """
    if not words:
        return html.escape(snippet)

    escaped = html.escape(snippet)
    for word in words:
        if not word:
            continue
        # 转义 word 中的正则特殊字符
        pattern = re.compile(re.escape(word), re.IGNORECASE)
        escaped = pattern.sub(
            lambda m: f'<mark class="bg-yellow-200 px-0.5 rounded">{m.group()}</mark>',
            escaped,
        )
    return escaped


# ──────────────────────────────────────────────
# 大文本读取
# ──────────────────────────────────────────────

def read_text_full(path: str | Path) -> str:
    """
    读取完整文本文件，UTF-8 → UTF-8-SIG → GBK 兜底。
    """
    p = Path(path)
    raw = p.read_bytes()
    for encoding in ("utf-8", "utf-8-sig", "gbk"):
        try:
            return raw.decode(encoding)
        except (UnicodeDecodeError, LookupError):
            continue
    raise ValueError("无法识别文本编码")


def read_text_segment(path: str | Path, offset: int = 0, length: int = 10000) -> str:
    """
    读取文本文件的指定片段（用于大文本分段显示）。
    offset: 字符偏移
    length: 读取字符数
    """
    full_text = read_text_full(path)
    return full_text[offset : offset + length]
```

## 4. 仓储层：排序 + 分页

`app/repositories/asset_repository.py`

修改 `list_assets` 函数签名和 SQL：

```python
def list_assets(
    conn,
    limit: int = 50,
    offset: int = 0,
    asset_type: str | None = None,
    order_by: str = "mtime",
    order_dir: str = "DESC",
) -> list[dict]:
    """
    列出资产，支持类型过滤、排序、分页。
    order_by: "title" / "mtime" / "size" / "type"
    order_dir: "ASC" / "DESC"
    """
    # 白名单校验，防 SQL 注入
    allowed_columns = {"title", "mtime", "size", "type"}
    if order_by not in allowed_columns:
        order_by = "mtime"
    if order_dir.upper() not in ("ASC", "DESC"):
        order_dir = "DESC"

    where_clause = ""
    params: list = []
    if asset_type:
        where_clause = "WHERE a.type = ?"
        params.append(asset_type)

    sql = f"""
        SELECT
            a.id,
            a.title,
            a.type,
            a.relative_path,
            a.absolute_path,
            a.size,
            a.mtime,
            a.parse_status,
            CASE WHEN EXISTS(
                SELECT 1 FROM artifacts ar
                WHERE ar.asset_id = a.id AND ar.kind = 'transcript'
            ) THEN 1 ELSE 0 END AS has_transcript,
            CASE WHEN EXISTS(
                SELECT 1 FROM artifacts ar
                WHERE ar.asset_id = a.id AND ar.kind = 'summary'
            ) THEN 1 ELSE 0 END AS has_summary,
            CASE WHEN EXISTS(
                SELECT 1 FROM artifacts ar
                WHERE ar.asset_id = a.id AND ar.kind = 'note'
            ) THEN 1 ELSE 0 END AS has_note,
            CASE WHEN EXISTS(
                SELECT 1 FROM artifacts ar
                WHERE ar.asset_id = a.id AND ar.kind = 'parsed'
            ) THEN 1 ELSE 0 END AS has_parsed,
            CASE WHEN EXISTS(
                SELECT 1 FROM artifacts ar
                WHERE ar.asset_id = a.id AND ar.kind = 'meta'
            ) THEN 1 ELSE 0 END AS has_meta
        FROM assets a
        {where_clause}
        ORDER BY a.{order_by} {order_dir}
        LIMIT ? OFFSET ?
    """
    params.extend([limit, offset])

    rows = conn.execute(sql, params).fetchall()
    return [dict(row) for row in rows]
```

from nicegui import run, ui

from app.database import get_conn
from app.services.search_service import search_filename, search_fulltext, search_vector
from app.state import get_db_path, state
from app.ui.layout import page_header, require_library
from app.utils import highlight_snippet, notify_error


@ui.page("/search")
def search_page():
    page_header("搜索", active_nav="search")
    if not require_library():
        return

    with ui.row().classes("items-center gap-3 mt-3 w-full"):
        query_input = ui.input(placeholder="输入搜索关键词…").classes("flex-1")
        search_btn = ui.button("全文搜索")
        filename_btn = ui.button("文件名搜索").props("outline")
        vector_btn = ui.button("语义搜索").props("outline color=teal")

    results_container = ui.column().classes("w-full mt-4")

    def render_results(results: list[dict], mode: str):
        results_container.clear()
        with results_container:
            if not results:
                ui.label("未找到相关结果。").classes("text-gray-500 mt-4")
                return
            ui.label(f"共 {len(results)} 条结果").classes(
                "text-sm text-gray-600 mb-2"
            )
            for result in results:
                with ui.card().classes("w-full p-3 mb-2"):
                    with ui.row().classes("items-center gap-2"):
                        ui.link(
                            result["asset_title"],
                            f"/assets/{result['asset_id']}",
                        ).classes("text-blue-600 font-medium")
                        # kind 徽章
                        ui.badge(result["kind"], color="grey").classes("text-xs")
                        # asset_type 徽章
                        ui.badge(
                            result["asset_type"], color="blue-grey"
                        ).classes("text-xs")
                        # 派生完成度徽章
                        _render_derived_badges(result)

                    # snippet 高亮渲染
                    snippet = result.get("snippet", "")
                    words = result.get("highlight_words", [])
                    if words and mode != "vector":
                        highlighted = highlight_snippet(snippet, words)
                        ui.html(
                            f'<p class="text-sm mt-2 whitespace-pre-wrap">{highlighted}</p>'
                        ).classes("w-full")
                    else:
                        ui.label(snippet).classes(
                            "text-sm mt-2 whitespace-pre-wrap text-gray-600"
                        )

                    # 路径
                    ui.label(result.get("relative_path", "")).classes(
                        "text-xs text-gray-400 mt-1"
                    )

    def _render_derived_badges(result: dict):
        """渲染派生完成度徽章（从搜索结果中获取）"""
        badges = result.get("derived_badges", {})
        if badges.get("has_transcript"):
            ui.badge("转录", color="green").classes("text-xs")
        if badges.get("has_summary"):
            ui.badge("总结", color="blue").classes("text-xs")
        if badges.get("has_note"):
            ui.badge("笔记", color="purple").classes("text-xs")
        if badges.get("has_parsed"):
            ui.badge("已解析", color="teal").classes("text-xs")
        if badges.get("has_meta"):
            ui.badge("元数据", color="grey-6").classes("text-xs")

    async def handle_fulltext():
        query = query_input.value.strip()
        if not query:
            ui.notify("请输入搜索关键词", type="warning")
            return
        try:
            results = await run.io_bound(search_fulltext, query)
            render_results(results, "fulltext")
        except Exception as exc:
            notify_error(exc)

    async def handle_filename():
        query = query_input.value.strip()
        if not query:
            ui.notify("请输入搜索关键词", type="warning")
            return
        try:
            results = await run.io_bound(search_filename, query)
            render_results(results, "filename")
        except Exception as exc:
            notify_error(exc)

    async def handle_vector():
        query = query_input.value.strip()
        if not query:
            ui.notify("请输入搜索关键词", type="warning")
            return
        try:
            results = await run.io_bound(search_vector, query)
            render_results(results, "vector")
        except Exception as exc:
            notify_error(exc)

    search_btn.on_click(handle_fulltext)
    filename_btn.on_click(handle_filename)
    vector_btn.on_click(handle_vector)
    query_input.on("keydown.enter", handle_fulltext)
```

> **注意**：`search_fulltext` / `search_filename` 返回结果中需要增加 `derived_badges` 字段。在 `search_service.py` 中，查询结果 JOIN `assets` 表时，同时 JOIN `artifacts` 聚合 `has_*` 字段（复用 `asset_repository` 中的子查询逻辑），或者在返回结果后批量查一次 `artifacts` 表。

                            "目录",

                notify_error(exc)

    sort_select.on_value_change(lambda: handle_filter_or_sort_change())
    prev_btn.on_click(handle_prev)
    next_btn.on_click(handle_next)

    load_assets()

def _render_text_section(asset: dict, artifact: dict):
    """渲染文本预览区域，支持展开全文和分段翻页"""
    abs_path = str(state.library_root / artifact["relative_path"])

    # 默认预览（前 2000 字符）
    preview_container = ui.column().classes("w-full mt-2")
    full_container = ui.column().classes("w-full mt-2")
    full_container.set_visibility(False)

    try:
        preview_text = read_text_preview(abs_path, max_chars=2000)
    except Exception:
        preview_text = "（无法读取预览）"

    with preview_container:
        ui.label(preview_text).classes(
            "text-sm whitespace-pre-wrap text-gray-700 bg-gray-50 p-3 rounded"
        )
        ui.label("预览已截断").classes("text-xs text-gray-400 mt-1")

    expand_btn = ui.button("展开全文").props("flat size=sm color=blue").classes(
        "mt-1"
    )
    collapse_btn = ui.button("收起").props("flat size=sm color=grey").classes("mt-1")
    collapse_btn.set_visibility(False)

    # 分段状态
    segment_state = {"offset": 0, "total_len": 0, "full_text": ""}

    async def handle_expand():
        try:
            text = await run.io_bound(read_text_full, abs_path)
        except Exception as exc:
            notify_error(exc)
            return

        segment_state["full_text"] = text
        segment_state["total_len"] = len(text)

        preview_container.set_visibility(False)
        full_container.set_visibility(True)
        expand_btn.set_visibility(False)
        collapse_btn.set_visibility(True)

        if len(text) <= FULL_TEXT_THRESHOLD:
            # 不需要分段，直接显示全文
            with full_container:
                full_container.clear()
                ui.label(text).classes(
                    "text-sm whitespace-pre-wrap text-gray-700 bg-gray-50 p-3 rounded max-h-[70vh] overflow-y-auto"
                )
        else:
            # 分段显示
            _render_segment(full_container, segment_state)

    def handle_collapse():
        preview_container.set_visibility(True)
        full_container.set_visibility(False)
        expand_btn.set_visibility(True)
        collapse_btn.set_visibility(False)

    expand_btn.on_click(handle_expand)
    collapse_btn.on_click(handle_collapse)


def _render_segment(container, segment_state: dict):
    """渲染分段文本 + 翻页控件"""
    container.clear()
    text = segment_state["full_text"]
    offset = segment_state["offset"]
    total_len = segment_state["total_len"]

    segment = text[offset : offset + SEGMENT_SIZE]
    current_page = offset // SEGMENT_SIZE + 1
    total_pages = (total_len + SEGMENT_SIZE - 1) // SEGMENT_SIZE

    with container:
        ui.label(segment).classes(
            "text-sm whitespace-pre-wrap text-gray-700 bg-gray-50 p-3 rounded max-h-[60vh] overflow-y-auto"
        )
        with ui.row().classes("items-center gap-3 mt-2"):
            seg_prev = ui.button("← 上一段").props("outline size=sm")
            ui.label(f"第 {current_page} / {total_pages} 段").classes(
                "text-sm text-gray-500"
            )
            seg_next = ui.button("下一段 →").props("outline size=sm")

            seg_prev.disabled = offset <= 0
            seg_next.disabled = offset + SEGMENT_SIZE >= total_len

            def go_prev():
                segment_state["offset"] = max(0, offset - SEGMENT_SIZE)
                _render_segment(container, segment_state)

            def go_next():
                segment_state["offset"] = offset + SEGMENT_SIZE
                _render_segment(container, segment_state)

            seg_prev.on_click(go_prev)
            seg_next.on_click(go_next)
```

def _get_derived_badges(conn, asset_id: str) -> dict:
    """查询某资产的派生完成度"""
    sql = """
        SELECT
            CASE WHEN EXISTS(
                SELECT 1 FROM artifacts WHERE asset_id = ? AND kind = 'transcript'
            ) THEN 1 ELSE 0 END AS has_transcript,
            CASE WHEN EXISTS(
                SELECT 1 FROM artifacts WHERE asset_id = ? AND kind = 'summary'
            ) THEN 1 ELSE 0 END AS has_summary,
            CASE WHEN EXISTS(
                SELECT 1 FROM artifacts WHERE asset_id = ? AND kind = 'note'
            ) THEN 1 ELSE 0 END AS has_note,
            CASE WHEN EXISTS(
                SELECT 1 FROM artifacts WHERE asset_id = ? AND kind = 'parsed'
            ) THEN 1 ELSE 0 END AS has_parsed,
            CASE WHEN EXISTS(
                SELECT 1 FROM artifacts WHERE asset_id = ? AND kind = 'meta'
            ) THEN 1 ELSE 0 END AS has_meta
    """
    row = conn.execute(sql, (asset_id, asset_id, asset_id, asset_id, asset_id)).fetchone()
    if row:
        return dict(row)
    return {}

## 16. 最近打开的 5 个知识库（M8 体验优化补充）

在首页提供"最近打开"卡片，记录并快捷切换最近使用过的知识库，避免每次重复粘贴路径。

### 16.1 数据层：recent_library_service

`app/services/recent_library_service.py`（新建）

```python
"""最近打开知识库记录（本地持久化，文件即数据）。

存储位置：<项目根>/data/recent_libraries.json
格式：[{ "path": "...", "opened_at": "ISO8601" }, ...]（最新在前，最多 MAX_RECENTS 条）
"""

MAX_RECENTS = 5
RECENTS_FILE = PROJECT_ROOT / "data" / "recent_libraries.json"


def _normalize(path_str: str) -> str:
    """规范化路径为绝对、解析符号链接的形式。"""
    return str(Path(path_str).expanduser().resolve())


def add_recent_library(path_str: str) -> None:
    """记录一个最近打开的知识库（去重、最新在前、超量截断到 MAX_RECENTS）。"""
    norm = _normalize(path_str)
    items = _read()
    items = [it for it in items if _normalize(it["path"]) != norm]
    items.insert(0, {"path": norm, "opened_at": datetime.now().isoformat(timespec="seconds")})
    items = items[:MAX_RECENTS]
    _write(items)


def get_recent_libraries() -> list[dict]:
    """返回最近打开且当前仍然存在的知识库列表（最新在前）。

    读取时剔除目录已不存在的条目，并回写清理后的列表。
    """
    items = _read()
    kept = [it for it in items if it.get("path") and Path(it["path"]).expanduser().is_dir()]
    if len(kept) != len(items):
        _write(kept)
    return kept


def clear_recents() -> None:
    """清空最近记录。"""
    if RECENTS_FILE.exists():
        RECENTS_FILE.unlink()
```

要点：
- `_read`/`_write` 容错：文件损坏或不存在时返回空列表，不会抛异常。
- 存储为本地 JSON（`data/recent_libraries.json`），不入 git（见 §16.4）。
- 最新在前、最多 5 条；同名路径自动去重并提到最前。
